Q2 count-only rerun:
```sql
BEGIN READ ONLY;
WITH expected(transport_id, mailbox) AS (
  VALUES
    ('transport-centurion', 'CEN'),
    ('transport-davinci', 'DV-shared'),
    ('transport-faith', 'DV-2'),
    ('transport-ayanda', 'DV-3'),
    ('transport-epick', 'DV-4'),
    ('transport-kmgidi', 'DV-5')
),
bounds AS (
  SELECT statement_timestamp() AS as_of
),
target AS (
  SELECT
    e.mailbox, m.id, m.message_id_header, m.transport_id, m.thread_id,
    m.subject, m.from_address, COALESCE(m.sent_at, m.created_at) AS sent_time,
    (mt.email IS NOT NULL AND mt.email <> ''
      AND lower(COALESCE(m.from_address, '')) LIKE '%' || lower(mt.email) || '%') AS sender_is_mailbox
  FROM message m
  JOIN expected e ON e.transport_id = m.transport_id
  JOIN mailbox_transport mt ON mt.id = m.transport_id
  CROSS JOIN bounds b
  WHERE m.direction = 'outbound'
    AND COALESCE(m.sent_at, m.created_at) >= b.as_of - interval '30 days'
    AND COALESCE(m.sent_at, m.created_at) < b.as_of
),
attachment_flags AS (
  SELECT a.message_id,
    bool_or(COALESCE(a.filename, '') ~* 'confirmation[[:space:]_-]*letter|confirmation[[:space:]_-]*no') AS file_confirmation
  FROM attachment a
  JOIN target t ON t.id = a.message_id
  GROUP BY a.message_id
),
letters AS (
  SELECT t.*
  FROM target t
  LEFT JOIN attachment_flags a ON a.message_id = t.id
  WHERE t.sender_is_mailbox
    AND (
      COALESCE(t.subject, '') ~* 'confirmation[[:space:]_-]*letter'
      OR COALESCE(t.subject, '') ~* 'confirmation[[:space:]_-]*no'
      OR COALESCE(a.file_confirmation, false)
    )
),
inbound_ranked AS (
  SELECT
    m.id, m.transport_id, m.thread_id, m.from_address, m.received_at, m.created_at,
    row_number() OVER (
      PARTITION BY m.transport_id, m.thread_id
      ORDER BY COALESCE(m.received_at, m.created_at), m.id
    ) AS rn
  FROM message m
  JOIN (
    SELECT DISTINCT transport_id, thread_id
    FROM letters
    WHERE thread_id IS NOT NULL
  ) lt ON lt.transport_id = m.transport_id AND lt.thread_id = m.thread_id
  WHERE m.direction = 'inbound'
),
first_inbound AS (
  SELECT id, transport_id, thread_id, from_address
  FROM inbound_ranked
  WHERE rn = 1
),
relevant_ids AS (
  SELECT id AS message_id FROM letters
  UNION
  SELECT id AS message_id FROM first_inbound
),
latest_payload AS (
  SELECT DISTINCT ON (me.message_id) me.message_id, me.payload
  FROM mail_extraction me
  JOIN relevant_ids r ON r.message_id = me.message_id
  WHERE me.payload IS NOT NULL AND jsonb_typeof(me.payload) = 'object'
  ORDER BY me.message_id, me.created_at DESC, me.id DESC
),
parsed AS (
  SELECT
    l.*, fi.id AS inbound_id, fi.from_address AS inbound_from,
    lp.payload AS letter_payload, ip.payload AS inbound_payload,
    regexp_match(
      COALESCE(l.subject, ''),
      'confirmation[[:space:]_-]*letter[_[:space:]]+(.+)[[:space:]]+[(]n[^0-9]*([0-9]+[-][0-9]+|[0-9]+)[)]',
      'i'
    ) AS letter_parts,
    regexp_match(
      COALESCE(l.subject, ''),
      'confirmation[[:space:]_-]*no[.]?[[:space:]]+([0-9]+[-][0-9]+|[0-9]+)[[:space:]]+(.+)$',
      'i'
    ) AS no_parts
  FROM letters l
  LEFT JOIN first_inbound fi
    ON fi.transport_id = l.transport_id AND fi.thread_id = l.thread_id
  LEFT JOIN latest_payload lp ON lp.message_id = l.id
  LEFT JOIN latest_payload ip ON ip.message_id = fi.id
),
parsed_fields AS (
  SELECT *,
    COALESCE(letter_parts[2], no_parts[1]) AS conf_no,
    COALESCE(letter_parts[1], no_parts[2]) AS subject_guest
  FROM parsed
),
payload_guest_records AS (
  SELECT
    x.message_id,
    NULLIF(trim(g.guest->>'name'), '') AS guest_name,
    NULLIF(trim(g.guest->>'confirmationNumber'), '') AS guest_conf,
    trim(regexp_replace(
      regexp_replace(lower(trim(g.guest->>'name')), '[[:punct:]]+', ' ', 'g'),
      '[[:space:]]+', ' ', 'g'
    )) AS guest_norm
  FROM latest_payload x
  CROSS JOIN LATERAL jsonb_array_elements(
    CASE WHEN jsonb_typeof(x.payload->'stays') = 'array'
      THEN x.payload->'stays' ELSE '[]'::jsonb END
  ) s(stay)
  CROSS JOIN LATERAL jsonb_array_elements(
    CASE WHEN jsonb_typeof(s.stay->'guests') = 'array'
      THEN s.stay->'guests' ELSE '[]'::jsonb END
  ) g(guest)
  WHERE NULLIF(trim(g.guest->>'name'), '') IS NOT NULL
),
candidate_guests AS (
  SELECT p.id AS letter_id, g.guest_name, g.guest_conf, g.guest_norm
  FROM parsed_fields p
  JOIN payload_guest_records g ON g.message_id = p.id
  UNION ALL
  SELECT p.id, g.guest_name, g.guest_conf, g.guest_norm
  FROM parsed_fields p
  JOIN first_inbound fi
    ON fi.transport_id = p.transport_id AND fi.thread_id = p.thread_id
  JOIN payload_guest_records g ON g.message_id = fi.id
),
guest_rollup AS (
  SELECT
    p.id,
    COUNT(DISTINCT g.guest_norm)
      FILTER (WHERE p.conf_no IS NOT NULL AND g.guest_conf = p.conf_no) AS matched_guest_n,
    MAX(g.guest_name)
      FILTER (WHERE p.conf_no IS NOT NULL AND g.guest_conf = p.conf_no) AS matched_guest_name,
    COUNT(DISTINCT g.guest_norm) AS any_guest_n,
    MAX(g.guest_name) AS any_guest_name
  FROM parsed_fields p
  LEFT JOIN candidate_guests g ON g.letter_id = p.id
  GROUP BY p.id
),
prepared AS (
  SELECT
    p.mailbox, p.id, p.message_id_header, p.inbound_id, p.subject_guest,
    COALESCE(
      p.subject_guest,
      CASE WHEN gr.matched_guest_n = 1 THEN gr.matched_guest_name END,
      CASE WHEN gr.any_guest_n = 1 THEN gr.any_guest_name END
    ) AS guest_name,
    NULLIF(trim(p.inbound_payload->'contact'->>'name'), '') AS payload_contact_name,
    CASE WHEN position('<' IN COALESCE(p.inbound_from, '')) > 0 THEN
      NULLIF(trim(translate(
        trim(regexp_replace(p.inbound_from, '[[:space:]]*<.*$', '')),
        chr(34), ''
      )), '')
    END AS sender_display_name,
    NULLIF(trim(p.inbound_payload->>'agency'), '') IS NOT NULL AS has_agency,
    (regexp_match(lower(COALESCE(p.inbound_from, '')), '@([[:alnum:].-]+)'))[1] AS from_domain,
    (regexp_match(lower(COALESCE(p.inbound_payload->'contact'->>'email', '')), '@([[:alnum:].-]+)'))[1] AS payload_contact_domain,
    gr.matched_guest_n, gr.any_guest_n
  FROM parsed_fields p
  LEFT JOIN guest_rollup gr ON gr.id = p.id
),
contacts AS (
  SELECT *,
    COALESCE(payload_contact_name, sender_display_name) AS contact_name,
    COALESCE(from_domain, payload_contact_domain) AS sender_domain,
    trim(regexp_replace(
      regexp_replace(lower(trim(guest_name)), '[[:punct:]]+', ' ', 'g'),
      '[[:space:]]+', ' ', 'g'
    )) AS guest_norm,
    trim(regexp_replace(
      regexp_replace(lower(trim(COALESCE(payload_contact_name, sender_display_name))), '[[:punct:]]+', ' ', 'g'),
      '[[:space:]]+', ' ', 'g'
    )) AS contact_norm,
    CASE
      WHEN inbound_id IS NULL THEN 'no-inbound'
      WHEN COALESCE(from_domain, payload_contact_domain) IS NULL THEN 'unknown'
      WHEN COALESCE(from_domain, payload_contact_domain) = ANY(ARRAY[
        'gmail.com', 'googlemail.com', 'yahoo.com', 'yahoo.co.za', 'yahoo.co.uk', 'ymail.com',
        'hotmail.com', 'outlook.com', 'live.com', 'msn.com', 'icloud.com', 'me.com',
        'aol.com', 'protonmail.com', 'proton.me', 'gmx.com', 'mail.com'
      ]) THEN 'personal-webmail'
      WHEN has_agency THEN 'agency-domain'
      ELSE 'company'
    END AS sender_class
  FROM prepared
),
analyzed AS (
  SELECT *,
    guest_norm IS NOT NULL AND contact_norm IS NOT NULL AND guest_norm <> contact_norm AS strict_name_mismatch,
    guest_norm IS NOT NULL AND (
      (contact_norm IS NOT NULL AND guest_norm <> contact_norm)
      OR sender_class = 'agency-domain'
    ) AS likely_booker_diff
  FROM contacts
),
summary AS (
  SELECT
    'class'::text AS scope, sender_class AS group_label,
    COUNT(*) AS n_letters,
    COUNT(DISTINCT COALESCE(message_id_header, id::text)) AS n_distinct_messages,
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL) AS n_first_inbound_joined,
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND guest_norm IS NOT NULL) AS n_joined_guest_available,
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND contact_norm IS NOT NULL) AS n_joined_contact_available,
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND guest_norm IS NOT NULL AND contact_norm IS NOT NULL) AS n_comparable,
    COUNT(*) FILTER (WHERE strict_name_mismatch) AS n_strict_name_mismatch,
    COUNT(*) FILTER (WHERE likely_booker_diff) AS n_likely_booker_diff,
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND subject_guest IS NOT NULL) AS n_joined_subject_guest,
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND subject_guest IS NULL AND guest_norm IS NOT NULL) AS n_joined_payload_guest_fallback,
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND payload_contact_name IS NOT NULL) AS n_contact_from_payload
  FROM analyzed
  GROUP BY sender_class
  UNION ALL
  SELECT
    'all', 'all',
    COUNT(*),
    COUNT(DISTINCT COALESCE(message_id_header, id::text)),
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL),
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND guest_norm IS NOT NULL),
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND contact_norm IS NOT NULL),
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND guest_norm IS NOT NULL AND contact_norm IS NOT NULL),
    COUNT(*) FILTER (WHERE strict_name_mismatch),
    COUNT(*) FILTER (WHERE likely_booker_diff),
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND subject_guest IS NOT NULL),
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND subject_guest IS NULL AND guest_norm IS NOT NULL),
    COUNT(*) FILTER (WHERE inbound_id IS NOT NULL AND payload_contact_name IS NOT NULL)
  FROM analyzed
)
SELECT
  scope, group_label, n_letters, n_distinct_messages, n_first_inbound_joined,
  n_joined_guest_available, n_joined_contact_available, n_comparable,
  n_strict_name_mismatch, n_likely_booker_diff, n_joined_subject_guest,
  n_joined_payload_guest_fallback, n_contact_from_payload
FROM summary
ORDER BY CASE scope WHEN 'all' THEN 0 ELSE 1 END, group_label;
ROLLBACK;
```
